5. Check if Decimal Number

Click to expand Code & Output

C Source Code:

#include <stdio.h>
#include <string.h>
int main() {
    char str[50];
    printf("Enter a number: ");
    scanf("%s", str);
    
    char *dot = strchr(str, '.');
    if (dot != NULL && *(dot + 1) != '\0' && strcmp(dot, ".0") != 0) {
        printf("It is a decimal number\n");
    } else {
        printf("It is not a decimal number\n");
    }
    return 0;
}

Expected Output:

Enter a number: 12.34
It is a decimal number

14. Months based on Day counts

Click to expand Code & Output

C Source Code:

#include <stdio.h>
int main() {
    int days;
    printf("Enter days (28, 29, 30, 31): ");
    scanf("%d", &days);
    
    if (days == 30) printf("Months: April, June, September, November\n");
    else if (days == 31) printf("Months: Jan, Mar, May, July, Aug, Oct, Dec\n");
    else if (days == 28 || days == 29) printf("Month: February\n");
    else printf("Invalid day input\n");
    
    return 0;
}

Months: April, June, September, November
