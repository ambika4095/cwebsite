1. Input and Print 50 Employee Details

#include < stdio.h>

struct employee {
    int id;
    char name[30], designation[30];
};

int main() {
    struct employee emp[50];
    int i;

    for(i=0; i<50; i++) {
        scanf("%d %s %s", &emp[i].id,
        emp[i].name, emp[i].designation);
    }

    printf("ID\tName\tDesignation\n");

    for(i=0; i<50; i++) {
        printf("%d\t%s\t%s\n", emp[i].id,
        emp[i].name, emp[i].designation);
    }

    return 0;
}

3. Print Name Characters with Employee ID

#include

struct employee {
    int id;
    char name[30], designation[30];
};

int main() {
    struct employee emp[50];
    int i, j;

    for(i=0; i<50; i++) {
        scanf("%d %s %s", &emp[i].id,
        emp[i].name, emp[i].designation);
    }

    printf("ID\tCharacter\n");

    for(i=0; i<50; i++) {
        for(j=0; emp[i].name[j]!='\0'; j++) {
            printf("%d\t%c\n", emp[i].id,
            emp[i].name[j]);
        }
    }

    return 0;
}

4. Student Details in Ascending Order of Roll Number

#include

struct student {
    int roll;
    char name[30], dob[20];
    float fee;
};

int main() {
    struct student s[50], temp;
    int i, j;

    for(i=0; i<50; i++) {
        scanf("%d %s %f %s",
        &s[i].roll, s[i].name,
        &s[i].fee, s[i].dob);
    }

    for(i=0; i<49; i++) {
        for(j=i+1; j<50; j++) {
            if(s[i].roll > s[j].roll) {
                temp=s[i];
                s[i]=s[j];
                s[j]=temp;
            }
        }
    }

    printf("Roll Name Fee DOB\n");

    for(i=0; i<50; i++) {
        printf("%d %s %.2f %s\n",
        s[i].roll, s[i].name,
        s[i].fee, s[i].dob);
    }

    return 0;
}
