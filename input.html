Answer: 
#include

    return 0;
}

ABCD

12. Enter Password Display Stars

Answer: 
#include
int main()
{
    char password[20];
    int i;
    printf("Enter password: ");
    scanf("%19s",password);

    for(i=0;password[i]!='\0';i++)
        printf("*");

    return 0;
}
Output: 
Enter password: ******

13. Enter Two Characters and Print ASCII

Answer: 
#include
int main()
{
    char a,b;
    printf("Enter two characters: ");
    scanf("%c %c",&a,&b);

    printf("ASCII of %c = %d\n",a,a);
    printf("ASCII of %c = %d",b,b);

    return 0;
}
Output: 
Enter two characters: A B
ASCII of A = 65
ASCII of B = 66
