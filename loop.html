Question 1. Enter a number from user and find binary of that number.

Solution

#include <stdio.h>
int main()
{
    int n, rem, bin=0, place=1;
    printf("Enter number: ");
    scanf("%d",&n);

    while(n>0)
    {
        rem=n%2;
        bin=bin+rem*place;
        place=place*10;
        n=n/2;
    }

    printf("Binary = %d",bin);
    return 0;
}

Output:

Enter number: 10
Binary = 1010

Question 12. Write a menu driven calculator using addition, subtraction, multiplication, division and modulo. Enter Q to quit.

Solution

#include <stdio.h>
int main()
{
    int a,b,ch;
    char choice;

    do
    {
        printf("\n1.Addition\n2.Subtraction\n3.Multiplication\n4.Division\n5.Modulo\n");
        printf("Enter choice: ");
        scanf("%d",&ch);

        if(ch>=1 && ch<=5)
        {
            printf("Enter two numbers: ");
            scanf("%d%d",&a,&b);
        }

        switch(ch)
        {
            case 1: printf("Answer = %d",a+b); break;
            case 2: printf("Answer = %d",a-b); break;
            case 3: printf("Answer = %d",a*b); break;
            case 4:
                if(b!=0) printf("Answer = %d",a/b);
                else printf("Cannot divide by zero");
                break;
            case 5:
                if(b!=0) printf("Answer = %d",a%b);
                else printf("Cannot divide by zero");
                break;
            default: printf("Invalid choice");
        }

        printf("\nPress Q to quit: ");
        scanf(" %c",&choice);

    } while(choice!='Q' && choice!='q');

    return 0;
}

Output:

1.Addition
2.Subtraction
3.Multiplication
4.Division
5.Modulo
Enter choice: 1
Enter two numbers: 10 5
Answer = 15
Press Q to quit: Q

Question 13. Enter a number and check whether it is palindrome or not.

Solution

{
    int n,temp,rev=0;
    printf("Enter number: ");

Question 24. Enter a number and check whether it is Armstrong or not.

Solution

#include <stdio.h>
int main()
{
    int n,temp,digit,sum=0;
    printf("Enter number: ");
    scanf("%d",&n);

    temp=n;
    while(n>0)
    {
        digit=n%10;
        sum=sum+digit*digit*digit;
        n=n/10;
    }

    if(temp==sum)
        printf("Armstrong Number");
    else
        printf("Not Armstrong Number");

    return 0;
}

Output:

Enter number: 153
Armstrong Number
